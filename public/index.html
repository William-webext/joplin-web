${htmlContent}
`); await dialogs.setButtons(dialogHandle, [{ id: 'ok', title: 'OK' }]); await dialogs.open(dialogHandle); }; // Funzione per recuperare ricorsivamente tutti i sotto-taccuini const getFolderTree = async (folderId: string): Promise => { const folders = []; const queue = [folderId]; while (queue.length > 0) { const currentId = queue.shift()!; try { const f = await joplin.data.get(['folders', currentId], { fields: ['id', 'title', 'parent_id'] }); folders.push(f); let page = 1; let hasMore = true; while (hasMore) { const res = await joplin.data.get(['folders', currentId, 'folders'], { fields: ['id'], page: page++ }); res.items.forEach((child: any) => queue.push(child.id)); hasMore = res.has_more; } } catch (e) { console.error("Error fetching folder details", e); } } return folders; }; const renderPanelContent = async () => { if (!panelHandle) return; const savedUrl = await joplin.settings.value('serverUrl'); const serverUrl = savedUrl.trim().replace(/\/$/, ''); let publishedFolders: any[] = []; try { const response = await fetch(`${serverUrl}/api/published-list`); const data = await response.json(); publishedFolders = data.folders || []; } catch (err: any) {} let itemsHtml = ''; if (publishedFolders.length === 0) { itemsHtml = `
ℹ️ No published notebooks found.
`; } else { publishedFolders.forEach((f) => { let visBadge = '🌍 Public'; if (f.visibility === 'private') visBadge = '🔒 Registered'; if (f.visibility === 'custom') visBadge = '🎯 Custom'; itemsHtml += `
📁 ${f.title} ${visBadge}
Web Notes: ${f.notesCount}
`; }); } const html = `

🌐 Web Publisher

${itemsHtml}
`; await joplin.views.panels.setHtml(panelHandle, html); }; const initPanelIfNeeded = async () => { if (!panelHandle) { panelHandle = await joplin.views.panels.create('webPublisherPanel'); joplin.views.panels.onMessage(panelHandle, async (msg: any) => { const savedUrl = await joplin.settings.value('serverUrl'); const foldersPayload = tree.map(f => ({ id: f.id, visibility: 'remove' })); await fetch(`${serverUrl}/api/publish`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ folders: foldersPayload }), }); await renderPanelContent(); } else if (msg.action === 'refresh') { await renderPanelContent(); } }); } }; await joplin.commands.register({ name: 'toggleWebPublisherPanel', label: '🌐 Toggle Web Publisher Panel', iconName: 'fas fa-globe', execute: async () => { await initPanelIfNeeded(); isPanelVisible = !isPanelVisible; if (panelHandle) { await joplin.views.panels.show(panelHandle, isPanelVisible); if (isPanelVisible) await renderPanelContent(); } }, }); // MAIN PUBLISH COMMAND (RECURSIVE) await joplin.commands.register({ name: 'publishNotebookToWeb', label: '🌐 Publish / Update / Remove from Web', iconName: 'fas fa-globe', execute: async (folderId?: string) => { let targetFolderId = folderId; if (!targetFolderId) { const selectedFolder = await joplin.workspace.selectedFolder(); if (selectedFolder) targetFolderId = selectedFolder.id; } if (!targetFolderId) return await showMessage('⚠️ Please select a notebook first.'); const savedUrl = await joplin.settings.value('serverUrl'); const serverUrl = savedUrl.trim().replace(/\/$/, ''); // Ottieni l'albero di tutti i sotto-taccuini const targetFolders = await getFolderTree(targetFolderId); if (targetFolders.length === 0) return await showMessage('⚠️ Error reading folder data.'); const rootFolder = targetFolders[0]; let users: any[] = []; let groups: any[] = []; let publishedList: any[] = []; try { const ugRes = await fetch(`${serverUrl}/api/users-and-groups`); const ugData = await ugRes.json(); users = ugData.users || []; groups = ugData.groups || []; const pubRes = await fetch(`${serverUrl}/api/published-list`); const pubData = await pubRes.json(); publishedList = pubData.folders || []; } catch (e) {} const currentPub = publishedList.find(p => p.id === targetFolderId); let currentVis = currentPub ? currentPub.visibility : 'public'; let currentAllowedUsers: string[] = currentPub ? (currentPub.allowedUsers || []) : []; let currentAllowedGroups: string[] = currentPub ? (currentPub.allowedGroups || []) : []; while (true) { let usersCheckboxes = users.map(u => { const isChecked = currentAllowedUsers.includes(u.email) ? 'checked' : ''; return ``; }).join(''); let groupsCheckboxes = groups.map(g => { const isChecked = currentAllowedGroups.includes(g.id) ? 'checked' : ''; return ``; }).join(''); const isCustomShow = currentVis === 'custom' ? 'block' : 'none'; await dialogs.setHtml(dialogHandle, `

Publish Notebook: "${rootFolder.title}"

${currentVis === 'public' ? '🌍 Visible to anyone on the web without login.' : ''} ${currentVis === 'private' ? '🔒 Visible to all authenticated registered users.' : ''} ${currentVis === 'custom' ? '🎯 Restricted to selected groups and users below.' : ''}
Authorized Groups:
${groupsCheckboxes || 'No groups available'}
Authorized Specific Users:
${usersCheckboxes || 'No users available'}
`); await dialogs.setButtons(dialogHandle, [{ id: 'ok', title: 'Confirm' }, { id: 'cancel', title: 'Cancel' }]); const result = await dialogs.open(dialogHandle); if (result.id === 'cancel') return; const formData = result.formData ? result.formData.publishForm : null; if (!formData) return; const visibility = formData.visibility; const allowedUsers: string[] = []; const allowedGroups: string[] = []; if (visibility === 'custom') { users.forEach(u => { if (formData[`user_${u.email}`]) allowedUsers.push(u.email); }); groups.forEach(g => { if (formData[`group_${g.id}`]) allowedGroups.push(g.id); }); if (allowedUsers.length === 0 && allowedGroups.length === 0) { currentVis = 'custom'; currentAllowedUsers = []; currentAllowedGroups = []; await showMessage('⚠️ Please select at least one authorized group or user for Custom Access.'); continue; } } try { let allNotes: any[] = []; if (visibility !== 'remove') { // Raccoglie tutte le note di TUTTI i sotto-taccuini for (const tf of targetFolders) { let page = 1; let hasMore = true; while (hasMore) { const response = await joplin.data.get(['folders', tf.id, 'notes'], { fields: ['id', 'parent_id', 'title', 'body', 'updated_time', 'user_updated_time'], page: page++, }); allNotes.push(...response.items); hasMore = response.has_more; } } } // Prepara l'array di taccuini da inviare al server const foldersPayload = targetFolders.map(tf => ({ id: tf.id, title: tf.title, visibility: visibility, allowedUsers, allowedGroups })); const payload = { folders: foldersPayload, notes: allNotes, }; const apiResponse = await fetch(`${serverUrl}/api/publish`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload), }); if (apiResponse.ok) { if (panelHandle && isPanelVisible) await renderPanelContent(); const subCountMsg = targetFolders.length > 1 ? `

Included ${targetFolders.length - 1} sub-notebook(s).` : ''; await showMessage(visibility === 'remove' ? `🗑️ Publication removed.${subCountMsg}` : `✅ Notebook "${rootFolder.title}" published successfully!${subCountMsg}`); } else { await showMessage(`❌ Operation failed.`); } } catch (error: any) { await showMessage(`❌ Error: ${error.message}`); } break; } }, }); await joplin.views.menuItems.create('publishNotebookItem', 'publishNotebookToWeb', MenuItemLocation.FolderContextMenu); await joplin.views.menuItems.create('toggleWebPublisherPanelItem', 'toggleWebPublisherPanel', MenuItemLocation.View); }, });
